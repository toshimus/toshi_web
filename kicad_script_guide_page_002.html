def create_fc_edge_connector():
    board = pcbnew.GetBoard()
    fp = pcbnew.FOOTPRINT(board)
    fp.SetReference("J1")
    fp.SetValue("FC_Edge_60Pin")
    
    try:
        fp.SetFPID(pcbnew.LIB_ID("", "FC_Edge_60Pin"))
    except Exception:
        pass
    
    fp_x, fp_y = 100.0, 133.5
    fp.SetPosition(pcbnew.VECTOR2I_MM(fp_x, fp_y))
    fp.SetOrientation(pcbnew.EDA_ANGLE(0, pcbnew.DEGREES_T))
    
    pitch = 2.54
    total_pins_per_side = 30
    pad_width = 1.2
    pad_length = 8.0
    
    lset_front = pcbnew.LSET()
    lset_front.AddLayer(pcbnew.F_Cu)
    lset_front.AddLayer(pcbnew.F_Mask)
    
    lset_back = pcbnew.LSET()
    lset_back.AddLayer(pcbnew.B_Cu)
    lset_back.AddLayer(pcbnew.B_Mask)
    
    for i in range(total_pins_per_side):
        pad = pcbnew.PAD(fp)
        pad.SetShape(pcbnew.PAD_SHAPE_RECT)
        pad.SetAttribute(pcbnew.PAD_ATTRIB_SMD)
        pad.SetLayerSet(lset_front)
        pad.SetSize(pcbnew.VECTOR2I_MM(pad_width, pad_length))
        
        x_pos = -((total_pins_per_side * pitch) / 2.0) + (i * pitch) + (pitch / 2.0)
        pad.SetPosition(pcbnew.VECTOR2I_MM(fp_x + x_pos, fp_y))
        pad.SetNumber(str(i + 1))
        fp.Add(pad)

    for i in range(total_pins_per_side):
        pad = pcbnew.PAD(fp)
        pad.SetShape(pcbnew.PAD_SHAPE_RECT)
        pad.SetAttribute(pcbnew.PAD_ATTRIB_SMD)
        pad.SetLayerSet(lset_back)
        pad.SetSize(pcbnew.VECTOR2I_MM(pad_width, pad_length))
        
        x_pos = -((total_pins_per_side * pitch) / 2.0) + (i * pitch) + (pitch / 2.0)
        pad.SetPosition(pcbnew.VECTOR2I_MM(fp_x + x_pos, fp_y))
        pad.SetNumber(str(i + 31))
        fp.Add(pad)

    board.Add(fp)

def fix_jp_board_outline_width():
    board = pcbnew.GetBoard()
    for drawing in list(board.GetDrawings()):
        if drawing.GetLayer() == pcbnew.Edge_Cuts:
            board.Delete(drawing)

    points = [
        (48, 45), (152, 45), (152, 115), (139, 115),
        (139, 138), (61, 138), (61, 115), (48, 115)
    ]
    
    for i in range(len(points)):
        p1 = points[i]
        p2 = points[(i + 1) % len(points)]
        line = pcbnew.PCB_SHAPE(board)
        line.SetShape(pcbnew.SHAPE_T_SEGMENT)
        line.SetLayer(pcbnew.Edge_Cuts)
        line.SetWidth(int(0.1 * 1000000))
        line.SetStart(pcbnew.VECTOR2I_MM(p1[0], p1[1]))
        line.SetEnd(pcbnew.VECTOR2I_MM(p2[0], p2[1]))
        board.Add(line)

【2段落】行ごとに詳しく説明

create_fc_edge_connector() 関数

ファミコン用の60ピン・エッジコネクタ（端子部）を、既存のライブラリに頼らずプログラムから動的に生成し、基板に配置する関数です。

  • board = pcbnew.GetBoard() 現在の基板データを取得します。
  • fp = pcbnew.FOOTPRINT(board) 基板上に配置するための新しい空のフットプリントオブジェクトを作成します。
  • fp.SetReference("J1") / fp.SetValue("FC_Edge_60Pin") フットプリントのリファレンスを「J1」、値を「FC_Edge_60Pin」に設定します。
  • try: ... except Exception: pass フットプリントの内部IDを設定しようと試みますが、エラーが起きても無視（pass）して処理を続行します。
  • fp_x, fp_y = 100.0, 133.5 フットプリント全体の基準となる中心座標（X: 100.0mm, Y: 133.5mm）を定義します。
  • fp.SetPosition(...) / fp.SetOrientation(...) フットプリントの配置座標と角度（0度）を設定します。
  • pitch / total_pins_per_side / pad_width / pad_length 端子の間距（2.54mm）、片面あたりのピン数（30ピン）、パッドの幅（1.2mm）と長さ（8.0mm）のパラメータを定義します。
  • lset_front = pcbnew.LSET() ... 表面（Front）用のレイヤーセットを作成し、銅箔層（F_Cu）とレジスト層（F_Mask）を割り当てます。
  • lset_back = pcbnew.LSET() ... 裏面（Back）用のレイヤーセットを作成し、銅箔層（B_Cu）とレジスト層（B_Mask）を割り当てます。
  • for i in range(total_pins_per_side): (1つ目のループ) 表面の30ピン分を生成するループ処理です。
  • pad = pcbnew.PAD(fp) 作成中のフットプリントに属する新しいパッド（端子）オブジェクトを生成します。
  • pad.SetShape(...) / pad.SetAttribute(...) パッドの形状を長方形（RECT）に、属性を表面実装（SMD）に設定します。
  • pad.SetLayerSet(lset_front) / pad.SetSize(...) パッドを表面レイヤーに設定し、定義した幅・サイズの寸法を適用します。
  • x_pos = ... 全体の中心を0とし、ピッチ（2.54mm）を基準にi番目のパッドの相対的なX座標を数学的に計算します。
  • pad.SetPosition(...) / pad.SetNumber(...) / fp.Add(pad) 計算した座標にパッドを配置し、ピン番号（1〜30）を割り当てて、フットプリントに追加します。
  • for i in range(total_pins_per_side): (2つ目のループ) 裏面の30ピン分を生成するループです。表面とほぼ同じ処理ですが、レイヤーが裏面（lset_back）になり、ピン番号が（31〜60）に設定されます。
  • board.Add(fp) 完成した60ピンのエッジコネクタ・フットプリントを基板全体に追加します。

fix_jp_board_outline_width() 関数

日本のファミコンカセット特有の段差がある形状（下部が細くなっている外形）の基板外形線を描画する関数です。

  • board = pcbnew.GetBoard() 現在の基板データを取得します。
  • for drawing in list(board.GetDrawings()): 基板上に描画されている図形（線やテキストなど）をすべて取得し、ループします。
  • if drawing.GetLayer() == pcbnew.Edge_Cuts: / board.Delete(drawing) もし図形のレイヤーが「Edge_Cuts（基板外形）」であれば、それを削除します。これにより古い外形線をリセットします。
  • points = [...] 新しい基板外形の角となる8つの頂点座標（X, Y）をリストとして定義します。特徴的なカセット端子部のくぼみを含む座標系です。
  • for i in range(len(points)): 頂点の数（8回）だけループ処理を行います。
  • p1 = points[i] / p2 = points[(i + 1) % len(points)] 始点（p1）と終点（p2）を取得します。最後は最初の点に戻るようモジュロ演算（%）を使用し、閉じた多角形を形成します。
  • line = pcbnew.PCB_SHAPE(board) 基板上に新しい図形オブジェクトを作成します。
  • line.SetShape(pcbnew.SHAPE_T_SEGMENT) 図形の形状を「線分（直線）」に設定します。
  • line.SetLayer(pcbnew.Edge_Cuts) 描画するレイヤーを「Edge_Cuts（基板外形）」に設定します。
  • line.SetWidth(int(0.1 * 1000000)) 線の太さを0.1mmに設定します。KiCadの内部単位（ナノメートル）に合わせるため、1000000を乗算しています。
  • line.SetStart(...) / line.SetEnd(...) 始点と終点の座標をミリメートル単位で設定します。
  • board.Add(line) 作成した線分を基板に追加します。これが8回繰り返され、外形線が完成します。

【3段落】今回使用した公式APIの解説

スクリプト内で使用されている、KiCad 7.0以降で推奨されるオブジェクトの動的生成に関連する主要なクラスとメソッドの解説です。

  • pcbnew.FOOTPRINT(board)
    ライブラリから読み込むのではなく、メモリ上で全く新しいフットプリントをゼロから生成するコンストラクタです。
  • pcbnew.PAD(footprint)
    特定のフットプリントに従属するパッド（ピン・端子）を生成します。形状（PAD_SHAPE_RECTなど）や実装タイプ（PAD_ATTRIB_SMDなど）を個別に設定し、最終的に footprint.Add(pad) で組み込みます。
  • pcbnew.LSET()
    レイヤーセット（Layer Set）を管理するクラスです。KiCadではパッドがどの層に存在するか（表面の銅箔、裏面のレジストなど）をビットマスクのようなセットで管理します。AddLayer() を使って複数層を同時に割り当てることが可能です。
  • pcbnew.PCB_SHAPE(board)
    基板上にグラフィカルな図形（線、円、多角形など）を描画するための汎用クラスです。SetShape() メソッドによって、線分（SHAPE_T_SEGMENT）や円（SHAPE_T_CIRCLE）などに変化します。
  • line.SetWidth(int(0.1 * 1000000)) の内部単位について
    KiCadのPython APIでは、幅や一部の長さ指定において、システム内部の基本単位（Internal Units: IU）を直接要求される場合があります。最新のKiCadでは 1 IU = 1 ナノメートル（10^-9 メートル）として扱われるため、0.1mm を指定する際は 1,000,000 を乗算して変換する手法が用いられます。
