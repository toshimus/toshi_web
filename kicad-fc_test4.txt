def create_fc_edge_connector():
    board = pcbnew.GetBoard()
    fp = pcbnew.FOOTPRINT(board)
    fp.SetReference("J1")
    fp.SetValue("FC_Edge_60Pin")
    
    try:
        fp.SetFPID(pcbnew.LIB_ID("", "FC_Edge_60Pin"))
    except Exception:
        pass
    
    fp_x, fp_y = 100.0, 133.5
    fp.SetPosition(pcbnew.VECTOR2I_MM(fp_x, fp_y))
    fp.SetOrientation(pcbnew.EDA_ANGLE(0, pcbnew.DEGREES_T))
    
    pitch = 2.54
    total_pins_per_side = 30
    pad_width = 1.2
    pad_length = 8.0
    
    lset_front = pcbnew.LSET()
    lset_front.AddLayer(pcbnew.F_Cu)
    lset_front.AddLayer(pcbnew.F_Mask)
    
    lset_back = pcbnew.LSET()
    lset_back.AddLayer(pcbnew.B_Cu)
    lset_back.AddLayer(pcbnew.B_Mask)
    
    for i in range(total_pins_per_side):
        pad = pcbnew.PAD(fp)
        pad.SetShape(pcbnew.PAD_SHAPE_RECT)
        pad.SetAttribute(pcbnew.PAD_ATTRIB_SMD)
        pad.SetLayerSet(lset_front)
        pad.SetSize(pcbnew.VECTOR2I_MM(pad_width, pad_length))
        
        x_pos = -((total_pins_per_side * pitch) / 2.0) + (i * pitch) + (pitch / 2.0)
        pad.SetPosition(pcbnew.VECTOR2I_MM(fp_x + x_pos, fp_y))
        pad.SetNumber(str(i + 1))
        fp.Add(pad)

    for i in range(total_pins_per_side):
        pad = pcbnew.PAD(fp)
        pad.SetShape(pcbnew.PAD_SHAPE_RECT)
        pad.SetAttribute(pcbnew.PAD_ATTRIB_SMD)
        pad.SetLayerSet(lset_back)
        pad.SetSize(pcbnew.VECTOR2I_MM(pad_width, pad_length))
        
        x_pos = -((total_pins_per_side * pitch) / 2.0) + (i * pitch) + (pitch / 2.0)
        pad.SetPosition(pcbnew.VECTOR2I_MM(fp_x + x_pos, fp_y))
        pad.SetNumber(str(i + 31))
        fp.Add(pad)

    board.Add(fp)

def pre_route_edge_shorts():
    board = pcbnew.GetBoard()
    j1 = board.FindFootprintByReference("J1")
    if not j1: return
    
    net_info = board.GetNetInfo()
    
    shorts = [
        (45, 46, "SOUND_IO", pcbnew.B_Cu),
        (48, 49, "CIRAM_CE", pcbnew.B_Cu)
    ]
    
    track_width = int(0.4 * 1000000)
    
    for p1, p2, netname, layer in shorts:
        net = net_info.GetNetItem(netname)
        if not net: continue
        
        pad1 = j1.FindPadByNumber(str(p1))
        pad2 = j1.FindPadByNumber(str(p2))
        
        if pad1 and pad2:
            pos1 = pad1.GetPosition()
            pos2 = pad2.GetPosition()
            
            top_y = pos1.y - int(5.5 * 1000000)
            
            pts = [
                pcbnew.VECTOR2I(pos1.x, pos1.y),
                pcbnew.VECTOR2I(pos1.x, top_y),
                pcbnew.VECTOR2I(pos2.x, top_y),
                pcbnew.VECTOR2I(pos2.x, pos2.y)
            ]
            
            for i in range(3):
                track = pcbnew.PCB_TRACK(board)
                track.SetStart(pts[i])
                track.SetEnd(pts[i+1])
                track.SetWidth(track_width)
                track.SetLayer(layer)
                track.SetNetCode(net.GetNetCode())
                board.Add(track)

def prepare_for_dsn_export():
    board = pcbnew.GetBoard()
    for track in list(board.GetTracks()):
        if not track.IsLocked():
            board.Remove(track)
            
    for zone in list(board.Zones()):
        board.Remove(zone)

def build_all():
    try:
        reset_board_components()
        fix_jp_board_outline_width()
        place_components()
        create_fc_edge_connector()
        add_wiring_nets()
        prepare_for_dsn_export()
        pre_route_edge_shorts()
        pcbnew.Refresh()
        print("DSN出力の準備が完了しました。切替スイッチ（SW1）が追加されました。")
    except Exception as e:
        print("スクリプト実行中にエラーが発生しました:")
        print(traceback.format_exc())

build_all()
